%!TEX program = xelatex
% !TeX encoding = UTF-8
%% ==========================================================
%%  重庆大学博士/硕士学位论文 LaTeX 模板（对齐 2023 版格式标准）
%%  使用前请先阅读 cquthesis.pdf 与 README，Happy TeXing!
%% ==========================================================
\documentclass%
	[type=doctor, bilinguallist=apart,]{cquthesis}
% 可用选项：
%   type=[bachelor|master|doctor]       必选，论文类型
%   liberalformat                        可选，仅本科，文学类标题格式
%   proffesionalmaster=[true|false]      可选，仅研究生，是否专业学位
%   printmode=[oneside|twoside|auto]     可选，打印方式，默认按页数自动判定
%   openany|openright                    可选，双面打印时每章是否右页起
%   bilinguallist=[off|combined|apart]   可选，图表索引双语方式
%   blindtrail                           可选，盲审模式（隐去姓名、致谢等）
%   draft                                可选，写作期草稿模式（不渲染图片）

\usepackage{cquthesis}    % 自定义宏包与命令写在这里，别堆在 main.tex
\graphicspath{{figures/}} % 图片默认目录

% 只编译部分章节（写作提速用，定稿时注释掉）
%\includeonly{contents/experiment,contents/analysis,}

\begin{document}

% ----------------------------------------------------------
% 前置部分
% ----------------------------------------------------------
% 封面与声明页不编页码，关闭 PDF 页面锚点，避免与正文第 1、2 页锚点重名
\hypersetup{pageanchor=false}
\input{contents/cover}   % 中英文题目、作者、导师等信息
\makecover               % 中文封面 + 英文题名页

\frontmatter
%% 独创性声明与使用授权书（2023 版：题名页之后、摘要之前）
% 已签字的扫描页可用：\cquauthpage[contents/authscan.pdf]
\cquauthpage
\clearpage
\hypersetup{pageanchor=true}
%% 中英文摘要与关键词
\makeabstract
%% 目录（需多次编译才会更新）
\tableofcontents
%% 插图和附表清单（可选）
\listoffigures
\listoffiguresEN
\listoftables
\listoftablesEN
%% 公式索引（可选）
\listofequations
\listofequationsEN
%% 主要符号对照表（可选）
\input{contents/denotation}

% ----------------------------------------------------------
% 主体部分
% 每章一个文件，便于管理与局部编译
% ----------------------------------------------------------
\mainmatter
\include{contents/introduction}
\include{contents/experiment}
\include{contents/analysis}
\include{contents/conclusion}

% ----------------------------------------------------------
% 后置部分：参考文献 -> 附录 -> 学位论文数据集 -> 致谢
% ----------------------------------------------------------
\backmatter

%% 参考文献（顺序编码制；至少引用一篇，否则会编译报错）
\bibliographystyle{cqunumerical}
\bibliography{ref/refs}

%% 附录：按 A、B、C 编号
\appendix
\include{contents/appendix}
%% 学位论文数据集（2023 版新增，附件 4；送审双盲版可留空）
\input{contents/dataset}

%% 致谢（2023 版：放在附录之后）
\include{contents/ack}

\end{document}
